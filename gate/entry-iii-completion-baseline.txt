def MatchingLogic.Missing : {S : MatchingLogic.Signature} → MatchingLogic.CanonicalCarrier S → Prop :=
fun {S} root => ∃ x, ∀ (world : MatchingLogic.GeneratedCarrier root), MatchingLogic.Pattern.var x ∉ ↑↑world
def MatchingLogic.CompletedCarrier : {S : MatchingLogic.Signature} → MatchingLogic.CanonicalCarrier S → Type :=
fun {S} root => { o // o = none → MatchingLogic.Missing root }
def MatchingLogic.CompletedCarrier.isStar : {S : MatchingLogic.Signature} →
  {root : MatchingLogic.CanonicalCarrier S} → MatchingLogic.CompletedCarrier root → Prop :=
fun {S} {root} point => ↑point = none
def MatchingLogic.completedEmbed : {S : MatchingLogic.Signature} →
  (root : MatchingLogic.CanonicalCarrier S) →
    MatchingLogic.GeneratedCarrier root → MatchingLogic.CompletedCarrier root :=
fun {S} root world => ⟨some world, ⋯⟩

def MatchingLogic.completedInterp : {S : MatchingLogic.Signature} →
  (root : MatchingLogic.CanonicalCarrier S) →
    (sigma : S.Sym) →
      (Fin (S.arity sigma) → MatchingLogic.CompletedCarrier root) → Set (MatchingLogic.CompletedCarrier root) :=
fun {S} root sigma inputs =>
  {output |
    ∃ components,
      (∀ (i : Fin (S.arity sigma)), inputs i = MatchingLogic.completedEmbed root (components i)) ∧
        ((∃ world,
            output = MatchingLogic.completedEmbed root world ∧
              world ∈ MatchingLogic.generatedInterp root sigma components) ∨
          output.isStar ∧ ∃ i, components i = MatchingLogic.generatedRoot root)}
@[reducible] def MatchingLogic.completedModel : {S : MatchingLogic.Signature} →
  MatchingLogic.CanonicalCarrier S → MatchingLogic.Model S :=
fun {S} root =>
  { carrier := MatchingLogic.CompletedCarrier root, nonempty := ⋯, interp := MatchingLogic.completedInterp root }

  {sigma : S.Sym} {inputs : Fin (S.arity sigma) → MatchingLogic.CompletedCarrier root}
  {output : MatchingLogic.CompletedCarrier root},
  output ∈ MatchingLogic.completedInterp root sigma inputs ↔
    ∃ components,
      (∀ (i : Fin (S.arity sigma)), inputs i = MatchingLogic.completedEmbed root (components i)) ∧
        ((∃ world,
            output = MatchingLogic.completedEmbed root world ∧
              world ∈ MatchingLogic.generatedInterp root sigma components) ∨
          output.isStar ∧ ∃ i, components i = MatchingLogic.generatedRoot root)

def MatchingLogic.completedValuation : {S : MatchingLogic.Signature} →
  (root : MatchingLogic.CanonicalCarrier S) → ℕ → MatchingLogic.CompletedCarrier root :=
fun {S} root x =>
  if h : ∃ world, MatchingLogic.Pattern.var x ∈ ↑↑world then MatchingLogic.completedEmbed root (Classical.choose h)
  else ⟨none, ⋯⟩

  (x : ℕ),
