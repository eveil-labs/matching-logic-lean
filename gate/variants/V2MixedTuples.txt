 ∀ (hne : C.Nonempty),
 ∀ star ∉ C,

 p ∈ (MatchingLogic.VariantMixed.cover₂ M C hne).denote ν ψ ↔

def MatchingLogic.VariantMixed.coverInterp₂ : {S : MatchingLogic.Signature} →
 (M : MatchingLogic.Model S) → (C : Set M.carrier) → (σ : S.Sym) → (Fin (S.arity σ) → ↑C × Bool) → Set (↑C × Bool) :=
fun {S} M C σ A => {p | ↑p.1 ∈ M.interp σ fun j => ↑(A j).1}
def MatchingLogic.VariantMixed.cover₂ : {S : MatchingLogic.Signature} →
 (M : MatchingLogic.Model S) → (C : Set M.carrier) → C.Nonempty → MatchingLogic.Model S :=
fun {S} M C hne => { carrier := ↑C × Bool, nonempty := ⋯, interp := MatchingLogic.VariantMixed.coverInterp₂ M C }
