def MatchingLogic.EntryIII.relLanguage : MatchingLogic.Signature → FirstOrder.Language :=
fun S => { Functions := fun x => Empty, Relations := fun n => { sigma // S.arity sigma + 1 = n } }
def MatchingLogic.EntryIII.relSym : {S : MatchingLogic.Signature} →
  (sigma : S.Sym) → (MatchingLogic.EntryIII.relLanguage S).Relations (S.arity sigma + 1) :=
fun {S} sigma => ⟨sigma, ⋯⟩
def MatchingLogic.EntryIII.Pattern.toFOAux.{u} : {S : MatchingLogic.Signature} →
  {Var : Type} →
    [DecidableEq Var] →
      MatchingLogic.Pattern S Var →
        {alpha : Type u} → (Var → alpha) → alpha → (MatchingLogic.EntryIII.relLanguage S).Formula alpha :=
fun {S} {Var} [DecidableEq Var] x x_1 x_2 x_3 =>
  MatchingLogic.Pattern.brecOn (motive := fun x =>
    (x : Type u) → (Var → x) → x → (MatchingLogic.EntryIII.relLanguage S).Formula x) x
    MatchingLogic.EntryIII.Pattern.toFOAux._f x_1 x_2 x_3
def MatchingLogic.EntryIII.Pattern.toFOFormula : {S : MatchingLogic.Signature} →
  {Var : Type} →
    [DecidableEq Var] → MatchingLogic.Pattern S Var → (MatchingLogic.EntryIII.relLanguage S).Formula (Option Var) :=
fun {S} {Var} [DecidableEq Var] p => MatchingLogic.EntryIII.Pattern.toFOAux p some none
def MatchingLogic.EntryIII.Pattern.toFOSentence : {S : MatchingLogic.Signature} →
  {Var : Type} →
    [DecidableEq Var] →
      MatchingLogic.Pattern S Var → ((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Sentence :=

  FirstOrder.Language.Formula.equivSentence (MatchingLogic.EntryIII.Pattern.toFOFormula p)
@[instance_reducible] def MatchingLogic.EntryIII.Model.toFOStructure : {S : MatchingLogic.Signature} →
  (M : MatchingLogic.Model S) → (MatchingLogic.EntryIII.relLanguage S).Structure M.carrier :=
fun {S} M =>
  { funMap := fun {n} a => isEmptyElim a,
    RelMap := fun {n} r xs =>
      have ys := fun i => xs (Fin.cast ⋯ i);
      ys 0 ∈ M.interp ↑r fun i => ys i.succ }
@[reducible] def MatchingLogic.EntryIII.Model.ofFOStructure : {S : MatchingLogic.Signature} →
  {Var : Type} →
    (A : Type) →
      [Nonempty A] →
        [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] → MatchingLogic.Model S :=
fun {S} {Var} A [Nonempty A] [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] =>
  { carrier := A, nonempty := ⋯,
    interp := fun sigma args =>
      {out | FirstOrder.Language.Structure.RelMap (Sum.inl (MatchingLogic.EntryIII.relSym sigma)) (Fin.cons out args)} }
def MatchingLogic.EntryIII.Model.ofFOValuation : {S : MatchingLogic.Signature} →
  {Var : Type} →
    (A : Type) → [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] → Var → A :=
fun {S} {Var} A [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] x =>
  ↑((MatchingLogic.EntryIII.relLanguage S).con (some x))
def MatchingLogic.EntryIII.Model.ofFOCurrent : {S : MatchingLogic.Signature} →
  {Var : Type} → (A : Type) → [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] → A :=
fun {S} {Var} A [((MatchingLogic.EntryIII.relLanguage S).withConstants (Option Var)).Structure A] =>
  ↑((MatchingLogic.EntryIII.relLanguage S).con none)
def MatchingLogic.EntryIII.Model.toFOParameters : {S : MatchingLogic.Signature} →
  {Var : Type} → (M : MatchingLogic.Model S) → (Var → M.carrier) → M.carrier → Option Var → M.carrier :=
fun {S} {Var} M rho u x =>
  match x with
  | none => u
  | some x => rho x
