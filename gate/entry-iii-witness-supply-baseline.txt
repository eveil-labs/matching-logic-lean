def MatchingLogic.InfiniteWitnessSupply : {S : MatchingLogic.Signature} → Set (MatchingLogic.Pattern S ℕ) → Prop :=
fun {S} Gamma =>
  ∀ {x : ℕ} {p : MatchingLogic.Pattern S ℕ},
    MatchingLogic.Pattern.ex x p ∈ Gamma →
      {y | (MatchingLogic.Pattern.ex x p).imp (MatchingLogic.Pattern.captureAvoidingSubst x y p) ∈ Gamma}.Infinite
def MatchingLogic.InfiniteFreshVariableSupply : {S : MatchingLogic.Signature} →
  Set (MatchingLogic.Pattern S ℕ) → Prop :=
fun {S} Gamma => {y | ∀ q ∈ Gamma, y ∉ MatchingLogic.FV q}.Infinite
