def MatchingLogic.canonicalInterp : {S : MatchingLogic.Signature} →
  (sigma : S.Sym) → (Fin (S.arity sigma) → MatchingLogic.CanonicalCarrier S) → Set (MatchingLogic.CanonicalCarrier S) :=
fun {S} sigma components =>
  {Gamma |
    ∀ (args : Fin (S.arity sigma) → MatchingLogic.Pattern S ℕ),
      (∀ (i : Fin (S.arity sigma)), args i ∈ ↑(components i)) → MatchingLogic.Pattern.app sigma args ∈ ↑Gamma}
@[reducible] def MatchingLogic.canonicalModel : {S : MatchingLogic.Signature} →
  MatchingLogic.CanonicalCarrier S → MatchingLogic.Model S :=
fun {S} root => { carrier := MatchingLogic.CanonicalCarrier S, nonempty := ⋯, interp := MatchingLogic.canonicalInterp }
