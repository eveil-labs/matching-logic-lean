@MatchingLogic.Model.denote_renameInjective : ∀ {S : MatchingLogic.Signature} {Var Var' : Type} [inst : DecidableEq Var]
  [inst_1 : DecidableEq Var'] (M : MatchingLogic.Model S) (f : Var → Var'),
  Function.Injective f →
    ∀ (rho : Var' → M.carrier) (p : MatchingLogic.Pattern S Var),
      M.denote rho (MatchingLogic.Pattern.rename f p) = M.denote (rho ∘ f) p
