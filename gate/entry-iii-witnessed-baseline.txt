fun {S} Gamma =>
  ∀ {x : ℕ} {p : MatchingLogic.Pattern S ℕ},
    MatchingLogic.Pattern.ex x p ∈ Gamma →
      ∃ y, (MatchingLogic.Pattern.ex x p).imp (MatchingLogic.Pattern.captureAvoidingSubst x y p) ∈ Gamma
