 (M : MatchingLogic.Model S) → Set M.carrier → Prop :=
fun {S} M C => ∀ ⦃u : M.carrier⦄, u ∈ C → ∀ ⦃v : M.carrier⦄, M.Step u v → v ∈ C
def MatchingLogic.AgreeOn : {S : MatchingLogic.Signature} →
 {Var : Type} → {M : MatchingLogic.Model S} → Set M.carrier → (Var → M.carrier) → (Var → M.carrier) → Prop :=
fun {S} {Var} {M} C ρ ρ' => ∀ (x : Var), ρ x = ρ' x ∧ ρ x ∈ C ∨ ρ x ∉ C ∧ ρ' x ∉ C

fun {S} M C σ A => {p | (∀ (j : Fin (S.arity σ)), (A j).2 = p.2) ∧ ↑p.1 ∈ M.interp σ fun j => ↑(A j).1}
