def MatchingLogic.coverInterp : {S : MatchingLogic.Signature} →
 (M : MatchingLogic.Model S) → (C : Set M.carrier) → (σ : S.Sym) → (Fin (S.arity σ) → ↑C × Bool) → Set (↑C × Bool) :=

def MatchingLogic.cover : {S : MatchingLogic.Signature} →
 (M : MatchingLogic.Model S) → (C : Set M.carrier) → C.Nonempty → MatchingLogic.Model S :=

def MatchingLogic.proj : {S : MatchingLogic.Signature} →
 (M : MatchingLogic.Model S) → (C : Set M.carrier) → M.carrier → Bool → ↑C × Bool → M.carrier :=
fun {S} M C star i p => if p.2 = i then ↑p.1 else star
def MatchingLogic.AgreeOn : {S : MatchingLogic.Signature} →
 {Var : Type} → {M : MatchingLogic.Model S} → Set M.carrier → (Var → M.carrier) → (Var → M.carrier) → Prop :=
fun {S} {Var} {M} C ρ ρ' => ∀ (x : Var), ρ x = ρ' x ∧ ρ x ∈ C ∨ ρ x ∉ C ∧ ρ' x ∉ C
