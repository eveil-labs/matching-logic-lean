def MatchingLogic.AlphaFreshWitnessed : {S : MatchingLogic.Signature} → Set (MatchingLogic.Pattern S ℕ) → Prop :=
fun {S} Gamma =>
  ∀ {x : ℕ} {p : MatchingLogic.Pattern S ℕ},
    MatchingLogic.Pattern.ex x p ∈ Gamma →
      ∃ x' p' y,
        (MatchingLogic.Pattern.ex x p).AlphaEq (MatchingLogic.Pattern.ex x' p') ∧
          y ∉ p'.allVars ∧
            (MatchingLogic.Pattern.ex x' p').imp (MatchingLogic.Pattern.captureAvoidingSubst x' y p') ∈ Gamma
MatchingLogic.fvBlocked_has_alpha_fresh_witness : ∃ x' p' y,
  (MatchingLogic.Pattern.ex 1 ((MatchingLogic.Pattern.var 1).and (MatchingLogic.Pattern.var 0))).AlphaEq
      (MatchingLogic.Pattern.ex x' p') ∧
    y ∉ p'.allVars ∧
      (MatchingLogic.Pattern.ex x' p').imp (MatchingLogic.Pattern.captureAvoidingSubst x' y p') ∈
        MatchingLogic.witnessCollapseTheory

@[reducible] def MatchingLogic.AlphaWitnessSig : MatchingLogic.Signature :=
{ Sym := MatchingLogic.AlphaWitnessSym, arity := fun x => 2 }
def MatchingLogic.pairArgs : MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ →
  MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ →
    Fin 2 → MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ :=
fun p q x =>
  match x with
  | ⟨0, isLt⟩ => p
  | ⟨1, isLt⟩ => q
MatchingLogic.pairArgs_zero : ∀ (p q : MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ),
  MatchingLogic.pairArgs p q 0 = p
MatchingLogic.pairArgs_one : ∀ (p q : MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ),
  MatchingLogic.pairArgs p q 1 = q
@[reducible] def MatchingLogic.alphaWitnessModel : MatchingLogic.Model MatchingLogic.AlphaWitnessSig :=
{ carrier := Bool, nonempty := ⋯, interp := fun x a => if a 0 = true ∧ a 1 = true then {true} else ∅ }
def MatchingLogic.alphaWitnessRho : ℕ → MatchingLogic.alphaWitnessModel.carrier :=
fun n => decide (n = 0)
def MatchingLogic.alphaWitnessTheory : Set (MatchingLogic.Pattern MatchingLogic.AlphaWitnessSig ℕ) :=
{p | true ∈ MatchingLogic.alphaWitnessModel.denote MatchingLogic.alphaWitnessRho p}
